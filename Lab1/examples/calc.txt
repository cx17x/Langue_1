method print_uint(n: int)
var digit: int;
begin
    if n >= 10 then begin
        print_uint(n / 10);
    end;
    digit := n % 10;
    send_byte('0' + digit);
end;

method print_int(n: int)
var value: int;
begin
    value := n;
    if value < 0 then begin
        send_byte('-');
        value := 0 - value;
    end;

    if value == 0 then
        send_byte('0')
    else
        print_uint(value);

    send_byte('\n');
end;

method main()
var ch: int;
    value: int;
    digit: int;
begin
    value := 0;

read_loop:
    ch := recv_byte();

    ; если перевод строки — конец ввода
    if ch == '\n' then begin
        print_int(value);
        return;
    end;

    ; digit = ch - '0'
    digit := ch - '0';

    ; value = value * 10 + digit
    value := value * 10 + digit;

    goto read_loop;
end;
