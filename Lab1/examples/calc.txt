method readInt(): int;
method writeInt(x: int);
method writeChar(c: char);

method main()
var op, a, b, res: int;
begin
    // читаем код операции: 0 = выход
    op := readInt();

    while op != 0 do
    begin
        a := readInt();
        b := readInt();
        res := 0;

        if op == 1 then
            res := a + b;
        else
            if op == 2 then
                res := a - b;
        else
            if op == 3 then
                res := a * b;
        else
            if op == 4 then
                res := a / b;
        else
            if op == 5 then
                res := a % b;
            else
                res := 0;

        writeInt(res);
        writeChar('\\n');

        op := readInt();
    end;
end;
