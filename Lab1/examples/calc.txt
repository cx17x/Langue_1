method read_byte(): byte;
method send_byte(ch: byte);
method print_int(n: int);

method calc_line()
var acc: int;
    op: byte;
    num: int;
    ch: byte;
    digit: int;
    have_op: bool;
    have_digit: bool;
begin
    acc := 0;
    have_op := false;
    have_digit := false;
    op := '+';
    num := 0;

    while 1 do begin
        ch := read_byte();

        if ch == 'q' then begin
            break;
        end else if ch == '\r' then begin
            // skip
        end else if ch == ' ' then begin
            // token separator: if we have a complete (op + number) => apply
            if have_op && have_digit then begin
                if op == '+' then begin
                    acc := acc + num;
                end else if op == '-' then begin
                    acc := acc - num;
                end else if op == '*' then begin
                    acc := acc * num;
                end else if op == '/' then begin
                    // protect from division by zero
                    if num == 0 then begin
                        send_byte('!');
                    end else begin
                        acc := acc / num;
                    end;
                end;

                // reset number
                num := 0;
                have_digit := false;
                have_op := false;
            end;
        end else if ch == '\n' then begin
            // end of line: apply pending token then print result and exit
            if have_op && have_digit then begin
                if op == '+' then begin
                    acc := acc + num;
                end else if op == '-' then begin
                    acc := acc - num;
                end else if op == '*' then begin
                    acc := acc * num;
                end else if op == '/' then begin
                    if num == 0 then begin
                        send_byte('!');
                    end else begin
                        acc := acc / num;
                    end;
                end;
            end;

            print_int(acc);
            break;

        end else if (ch == '+') || (ch == '-') || (ch == '*') || (ch == '/') then begin
            // start new operation token
            op := ch;
            num := 0;
            have_op := true;
            have_digit := false;

        end else begin
            // digit
            digit := ch - '0';
            // (optional) check 0..9
            if (digit >= 0) && (digit <= 9) then begin
                num := num * 10;
                num := num + digit;
                have_digit := true;
            end else begin
                // ignore unknown char
            end;
        end;
    end;
end;

method main()
begin
    // one line calculator; you can call in a loop if you want multiple lines
    calc_line();
end;
